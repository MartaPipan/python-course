

🛠️ 3. Повний практичний приклад: як додати правила та перевірки?

Оскільки @dataclass сам генерує __init__, як нам додати перевірку (валідацію)? Наприклад, перевірити, щоб назва книги не була порожньою?
Для цього в @dataclass є спеціальний магічний метод — __post_init__(self). Він автоматично запускається одразу після того, як датаклас створив та заповнив усі поля.


from dataclasses import dataclass, field

@dataclass
class Book:
    title: str
    author: str
    tags: list[str] = field(default_factory=list) # Свіжий список для кожної книги [1]

    # Додаємо складні правила та перевірки після ініціалізації
    def __post_init__(self):
        if not self.title.strip():
            raise ValueError("Назва книги не може бути порожньою!")
        if not self.author.strip():
            raise ValueError("Автор має бути вказаний!")


# ==========================================
# 🎮 ТЕСТУВАННЯ
# ==========================================
if __name__ == "__main__":
    # 1. Успішне створення
    book1 = Book("Місто", "В. Підмогильний")
    print(book1) # Авто-repr працює ідеально!

    # 2. Перевірка незалежності списків
    book2 = Book("Тигролови", "І. Багряний")
    book2.tags.append("класика") # Додаємо тег ТІЛЬКИ другій книзі

    print(f"Теги книги 1: {book1.tags}") # Залишився порожнім: []
    print(f"Теги книги 2: {book2.tags}") # Має тег: ['класика']

    # 3. Спроба порушити правила (валідація в __post_init__)
    try:
        broken_book = Book("   ", "Хтось")
    except ValueError as e:
        print(f"🔒 Розумне проєктування захистило модель! Помилка: {e}")
